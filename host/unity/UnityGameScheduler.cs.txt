// 作用：用 Unity scaled 游戏时间为 Cont/fx 提供
//   host.now() / host.schedule(delay, cb) / host.cancel(id)
// 并保证 Lua 回调在主线程执行。
// 可选：ScheduleLuaReal → host.schedule_real（仅 fx.wait_real；默认业务禁用）。
//
// 推荐：scaled 时间（WaitForSeconds / Time.time），这样
//   Time.timeScale = 0  时 wait 自然暂停；恢复后继续。
// Pause 菜单：设 timeScale=0 即可，不必在本类再维护 pause 标志。
//
// 本文件是模板：Lua 调用处标有 TODO，按 xLua / tolua / slua 补全。
// 不要把它当成可直接编译的完整插件。
//
// ── xLua 快速路径（拷贝清单见 host/unity/README.md）──
//   1. 本文件改名为 Assets/.../UnityGameScheduler.cs，挂到 DontDestroy 物体
//   2. Awake/Start 里调用 InjectIntoLua(luaenv)
//   3. Lua：local api = require("FxUnityBootstrap").from_global("UnityHost")
//           api.run(fx.wait(1.0) >> ...)
//   4. ScheduleLua / ScheduleLuaReal 的 TODO 换成 (cb as LuaFunction).Action()

using System;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// Cont/fx 游戏时间后端（scaled）。挂到常驻物体上。
/// </summary>
public class UnityGameScheduler : MonoBehaviour
{
    public static UnityGameScheduler Instance { get; private set; }

    private int _nextId = 1;
    private readonly Dictionary<int, Coroutine> _running = new Dictionary<int, Coroutine>();
    private readonly Queue<Action> _mainThread = new Queue<Action>();
    private readonly object _queueLock = new object();

    void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
        DontDestroyOnLoad(gameObject);
    }

    void Update()
    {
        // 主线程出队：到期 timer 只入队，真正碰 Lua 在这里
        while (true)
        {
            Action a = null;
            lock (_queueLock)
            {
                if (_mainThread.Count > 0)
                    a = _mainThread.Dequeue();
            }
            if (a == null) break;
            try { a(); }
            catch (Exception e) { Debug.LogException(e); }
        }
    }

    /// <summary>当前 scaled 游戏时间（秒），对齐 Time.time。</summary>
    public double Now()
    {
        return Time.time;
    }

    /// <summary>
    /// delay 秒后（scaled）在主线程调用 onDue。
    /// 返回 id，供 Cancel。
    /// </summary>
    public int Schedule(float delay, Action onDue)
    {
        int id = _nextId++;
        Coroutine co = StartCoroutine(CoDelay(id, delay, onDue, realtime: false));
        _running[id] = co;
        return id;
    }

    /// <summary>
    /// delay 秒后（墙钟 / unscaled）在主线程调用 onDue。
    /// 仅供 Lua host.schedule_real / fx.wait_real；游戏逻辑请继续用 Schedule。
    /// </summary>
    public int ScheduleReal(float delay, Action onDue)
    {
        int id = _nextId++;
        Coroutine co = StartCoroutine(CoDelay(id, delay, onDue, realtime: true));
        _running[id] = co;
        return id;
    }

    /// <summary>取消尚未触发的调度；迟到回调会被跳过。</summary>
    public void Cancel(int id)
    {
        if (_running.TryGetValue(id, out Coroutine co))
        {
            if (co != null) StopCoroutine(co);
            _running.Remove(id);
        }
    }

    private IEnumerator CoDelay(int id, float delay, Action onDue, bool realtime)
    {
        if (delay > 0f)
        {
            if (realtime)
                yield return new WaitForSecondsRealtime(delay);
            else
                yield return new WaitForSeconds(delay); // scaled：受 timeScale 影响
        }
        else
        {
            yield return null;
        }

        _running.Remove(id);
        EnqueueMainThread(() =>
        {
            // 若 Cancel 已拆掉，上面 Remove 后不会进到这里的竞态由 Lua 侧 handle.cancelled 再挡一层
            onDue?.Invoke();
        });
    }

    private void EnqueueMainThread(Action a)
    {
        lock (_queueLock) { _mainThread.Enqueue(a); }
    }

    // -------------------------------------------------------------------------
    // 以下：把本组件暴露给 Lua（xLua 示例可直接改 TODO 后用）
    // -------------------------------------------------------------------------

    /// <summary>
    /// 向全局注入 UnityHost = { now, schedule, cancel [, schedule_real] }。
    /// 之后 Lua：require("FxUnityBootstrap").from_global() → api.run(...)
    /// </summary>
    public void InjectIntoLua(/* XLua.LuaEnv env */)
    {
        // TODO(xLua)：取消注释并换成你的 LuaEnv 变量名。
        //
        // env.DoString(@"
        //   UnityHost = UnityHost or {}
        //   local U = CS.UnityGameScheduler.Instance
        //   function UnityHost.now()
        //     return U:Now()
        //   end
        //   function UnityHost.schedule(delay, cb)
        //     return U:ScheduleLua(delay, cb)
        //   end
        //   function UnityHost.cancel(id)
        //     U:Cancel(id)
        //   end
        //   -- 可选：仅当确需 fx.wait_real（网络/SDK）时打开
        //   function UnityHost.schedule_real(delay, cb)
        //     return U:ScheduleLuaReal(delay, cb)
        //   end
        // ");
        //
        // 等价「不 DoString、纯 C# Push」示意：
        //   env.Global.Set("UnityHost", env.NewTable());
        //   // 再为表 Set("now", ...) 三个/四个委托 —— 视 xLua 版本 API 而定。
        //
        // ScheduleLua / ScheduleLuaReal 必须把 LuaFunction 包成 Action，并在主线程 Call。
    }

    /// <summary>
    /// 供 Lua 调用的 schedule（scaled）：把 Lua 回调封成 Action，到期主线程执行。
    /// </summary>
    public int ScheduleLua(float delay, object luaCallback)
    {
        return Schedule(delay, () => InvokeLuaCallback(luaCallback, "ScheduleLua"));
    }

    /// <summary>
    /// 供 Lua 调用的墙钟 schedule（unscaled）。对应 host.schedule_real。
    /// </summary>
    public int ScheduleLuaReal(float delay, object luaCallback)
    {
        return ScheduleReal(delay, () => InvokeLuaCallback(luaCallback, "ScheduleLuaReal"));
    }

    private static void InvokeLuaCallback(object luaCallback, string tag)
    {
        if (luaCallback == null) return;
        // TODO(xLua)：
        //   var fn = luaCallback as XLua.LuaFunction;
        //   fn?.Action();          // 无参回调；或 fn.Call()
        //   // 不要在非主线程缓存/调用 LuaFunction；本类已保证在 Update 出队后调用。
        // TODO(tolua / slua)：换成对应的 LuaFunction.Call。
        Debug.LogWarning("UnityGameScheduler." + tag + ": wire luaCallback for your Lua host");
    }
}
